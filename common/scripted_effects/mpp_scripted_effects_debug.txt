@debug_all = 0
@debug_error = 1
@debug_info = 2
@debug_configs = 3

mpp_debug = {
    [[LEVEL]
        set_variable = { which = debugLevel value = $LEVEL$ }
        if = {
            limit = { check_variable_arithmetic = { which = debugLevel value = 1 } }
            mpp_enble_debug_error = yes
        }
        else_if = {
            limit = { check_variable_arithmetic = { which = debugLevel value <= 2 } }
            mpp_enble_debug_error = yes
            mpp_enble_debug_info = yes
        }
        else = {
            mpp_enble_debug_all = yes
        }
    ]
    [[!LEVEL]
        mpp_enble_debug_all = yes
    ]
}

mpp_enble_debug = {
    if = {
        limit = { mpp_is_debug_active = no }
        log_error = "🐞 - MPP - DEBUG ACTIVO"
    }
    set_global_flag = mpp_flag_debug
}

mpp_enble_debug_all = {
    if = {
        limit = { mpp_is_debug_active = no }

    if = {
        limit = { OR = { is_scope_valid = no NOR = { is_scope_type = leader is_scope_type = pop } } }
        mpp_log_error_invalid_scope = yes
    }
}

}
