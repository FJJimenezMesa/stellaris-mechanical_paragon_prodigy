mpp_debug_check_scope_pop_or_leader = {
    if = {
        limit = {
            NOT = {
                OR = {
                    is_scope_type = pop
                    is_scope_type = leader
                    is_scope_valid = yes
                }
            }
        }
        mpp_debug_log_error_invalid_scope = yes
    }
}
