mpp_debug_log_error_invalid_scope = {
    if = {
        limit = { mpp_is_debug_active = yes }
        if = { limit = { is_scope_valid = no } log_error = "❌ - MPP - No valid scope" }
        else_if = { limit = { is_scope_type = none } log_error = "❌ - MPP - Current scope: none" }
        else_if = { limit = { is_scope_type = country } log_error = "❌ - MPP - Current scope: country" }
        else_if = { limit = { is_scope_type = sector } log_error = "❌ - MPP - Current scope: sector" }
        else_if = { limit = { is_scope_type = galactic_object } log_error = "❌ - MPP - Current scope: galactic_object" }
        else_if = { limit = { is_scope_type = megastructure } log_error = "❌ - MPP - Current scope: megastructure" }
        else_if = { limit = { is_scope_type = ambient_object } log_error = "❌ - MPP - Current scope: ambient_object" }
        else_if = { limit = { is_scope_type = planet } log_error = "❌ - MPP - Current scope: planet" }
        else_if = { limit = { is_scope_type = deposit } log_error = "❌ - MPP - Current scope: deposit" }
        else_if = { limit = { is_scope_type = archaeological_site } log_error = "❌ - MPP - Current scope: archaeological_site" }
        else_if = { limit = { is_scope_type = army } log_error = "❌ - MPP - Current scope: army" }
        else_if = { limit = { is_scope_type = pop } log_error = "❌ - MPP - Current scope: pop" }
        else_if = { limit = { is_scope_type = pop_faction } log_error = "❌ - MPP - Current scope: pop_faction" }
        else_if = { limit = { is_scope_type = species } log_error = "❌ - MPP - Current scope: species" }
        else_if = { limit = { is_scope_type = leader } log_error = "❌ - MPP - Current scope: leader" }
        else_if = { limit = { is_scope_type = ship } log_error = "❌ - MPP - Current scope: ship" }
        else_if = { limit = { is_scope_type = fleet } log_error = "❌ - MPP - Current scope: fleet" }
        else_if = { limit = { is_scope_type = federation } log_error = "❌ - MPP - Current scope: federation" }
        else_if = { limit = { is_scope_type = war } log_error = "❌ - MPP - Current scope: war" }
        else_if = { limit = { is_scope_type = first_contact } log_error = "❌ - MPP - Current scope: first_contact" }
        else_if = { limit = { is_scope_type = espionage_operation } log_error = "❌ - MPP - Current scope: espionage_operation" }
        else = { log_error = "❌ - MPP - Current scope: UNKNOWN (not matched)" }
    }
}

mpp_debug_check_scope_pop_or_leader = {
    if = {
        limit = { OR = { is_scope_valid = no NOR = { is_scope_type = leader is_scope_type = pop } } }
        mpp_debug_log_error_invalid_scope = yes
    }
}

mpp_debug_check_scope_leader = {
    if = {
        limit = { OR = { is_scope_valid = no  NOT = { is_scope_type = leader } } }
        mpp_debug_log_error_invalid_scope = yes
    }
}
