@script_min_distance = -50
@script_max_distance = 299

mpp_set_location = {
    optimize_memory

    [[TARGET]
        if = {
            limit = { exists = event_target:$TARGET$ }
            # Cambiamos el scope donde hacer set_location
            event_target:$TARGET$ = {
                inline_script = {
                    script = mpp_i_s_set_location
                    EFFECT = mpp_set_location
                    LOCATION = $LOCATION|@mpp_no$
                    LOCATION_TARGET = $LOCATION_TARGET|@mpp_no$
                    DISTANCE = $DISTANCE|@mpp_no$
                    DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|@mpp_no$
                    MIN_DISTANCE = $MIN_DISTANCE|0$
                    MAX_DISTANCE = $MAX_DISTANCE|0$
                    IN_ORBIT = $IN_ORBIT|@mpp_no$
                }
            }
        }
        else = {
            mpp_log_effect_error = {
                EFFECT = mpp_set_location TEXT = "TARGET $TARGET$ no válido"
            }
        }
    ]

    [[!TARGET]
        inline_script = {
            script = mpp_i_s_set_location
            EFFECT = mpp_set_location
            LOCATION = $LOCATION|@mpp_no$
            LOCATION_TARGET = $LOCATION_TARGET|@mpp_no$
            DISTANCE = $DISTANCE|@mpp_no$
            DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|@mpp_no$
            MIN_DISTANCE = $MIN_DISTANCE|0$
            MAX_DISTANCE = $MAX_DISTANCE|0$
            IN_ORBIT = $IN_ORBIT|@mpp_no$
        }
    ]
}
