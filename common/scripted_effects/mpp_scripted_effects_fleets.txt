# !rev no se pueden crear flotas dinámicamente falla owner en events/paragon_events_3
mpp_add_fleet = {
    optimize_memory

    create_fleet = {
        [[FLEET_NAME] name = $FLEET_NAME$ ]

        effect = {
            inline_script = {
                script = fleets/mpp_i_s_fleet_config
                EFFECT = mpp_add_fleet
                FLEET_NAME = @mpp_no # !Rev Ya puesto. No admite variables de texto, da error de scope
                FLEET_OWNER = $FLEET_OWNER|@mpp_no$
                FLEET_OWNER_TARGET = @mpp_no # !Rev No admite variables de texto, da error de scope
                # -- Location config -- #
                LOCATION = $LOCATION|@mpp_no$
                LOCATION_TARGET = @mpp_no # !Rev No admite variables de texto, da error de scope
                DISTANCE = $DISTANCE|@mpp_no$
                DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|@mpp_no$
                MIN_DISTANCE = $MIN_DISTANCE|0$
                MAX_DISTANCE = $MAX_DISTANCE|0$
                ANGLE = $ANGLE|@mpp_random$
                # -- Aggro config -- #
                FLEET_STANCE = $FLEET_STANCE|@mpp_fleet_stance_passive$
                AGGRO_RANGE = $AGGRO_RANGE|150$
                AGGRO_FROM_RETURN_POINT = @mpp_no
                # -- Fleet Ships config -- #
                FLEET_DESIGN = @mpp_no # !Rev No admite variables de texto, da error de scope
                FLEET_DESIGN_TYPE = $FLEET_DESIGN_TYPE|@mpp_fleet_design_base$
                SHIP_1 = $SHIP_1|0$
                SHIP_1_NAME = @mpp_random # !Rev No admite variables de texto, da error de scope
                SHIP_1_DESIGN = @mpp_random # !Rev No admite variables de texto, da error de scope
                SHIP_2 = $SHIP_2|0$
                SHIP_2_NAME = @mpp_random # !Rev No admite variables de texto, da error de scope
                SHIP_2_DESIGN =@mpp_random # !Rev No admite variables de texto, da error de scope
                SHIP_3 = $SHIP_3|0$
                SHIP_3_NAME = @mpp_random # !Rev No admite variables de texto, da error de scope
                SHIP_4_DESIGN =@mpp_random # !Rev No admite variables de texto, da error de scope
                SHIP_4 = $SHIP_4|0$
                SHIP_4_NAME = @mpp_random # !Rev No admite variables de texto, da error de scope
                SHIP_4_DESIGN = @mpp_random # !Rev No admite variables de texto, da error de scope
                # -- Fleet Actions -- #
                MOVE_TO_TARGET = @mpp_no # !Rev No admite variables de texto, da error de scope
                MOVE_TO_PLANET_FLAG = @mpp_no # !Rev No admite variables de texto, da error de scope
                PATROL_PLANET_FLAG_1 = @mpp_no # !Rev No admite variables de texto, da error de scope
                PATROL_PLANET_FLAG_2 = @mpp_no # !Rev No admite variables de texto, da error de scope
                PATROL_PLANET_FLAG_3 =@mpp_no # !Rev No admite variables de texto, da error de scope
                PATROL_PLANET_FLAG_4 = @mpp_no # !Rev No admite variables de texto, da error de scope
                PATROL_PLANET_FLAG_4 = @mpp_no # !Rev No admite variables de texto, da error de scope
            }
        }
    }
}

# !rev Para evitar problemas de scope hay que inicializar variables de text antes de llamada
mpp_fleet_config = {
    optimize_memory

    if = {
        limit = { is_scope_type = fleet }

        hidden_effect = {
            set_variable = { which = base value = @mpp_yes } # !rev Para evitar problemas de scope

            inline_script = {
                script = fleets/mpp_i_s_fleet_config
                EFFECT = mpp_add_fleet
                FLEET_NAME = $FLEET_NAME|@mpp_no$
                FLEET_OWNER = $FLEET_OWNER|@mpp_no$
                FLEET_OWNER_TARGET = $FLEET_OWNER_TARGET|@mpp_no$
                # -- Location config -- #
                LOCATION = $LOCATION|@mpp_no$
                LOCATION_TARGET = $LOCATION_TARGET|@mpp_no$
                DISTANCE = $DISTANCE|@mpp_no$
                DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|@mpp_no$
                MIN_DISTANCE = $MIN_DISTANCE|0$
                MAX_DISTANCE = $MAX_DISTANCE|0$
                ANGLE = $ANGLE|@mpp_random$
                # -- Aggro config -- #
                FLEET_STANCE = $FLEET_STANCE|@mpp_fleet_stance_passive$
                AGGRO_RANGE = $AGGRO_RANGE|150$
                AGGRO_FROM_RETURN_POINT = @mpp_no
                # -- Fleet Ships config -- #
                FLEET_DESIGN = $FLEET_DESIGN|base$
                FLEET_DESIGN_TYPE = $FLEET_DESIGN_TYPE|@mpp_fleet_design_base$
                SHIP_1 = $SHIP_1|0$
                SHIP_1_NAME = $SHIP_1_NAME|@mpp_random$
                SHIP_1_DESIGN = $SHIP_1_DESIGN|@mpp_random$
                SHIP_2 = $SHIP_2|0$
                SHIP_2_NAME = $SHIP_2_NAME|@mpp_random$
                SHIP_2_DESIGN = $SHIP_2_DESIGN|@mpp_random$
                SHIP_3 = $SHIP_3|0$
                SHIP_3_NAME = $SHIP_3_NAME|@mpp_random$
                SHIP_4_DESIGN = $SHIP_4_DESIGN|@mpp_random$
                SHIP_4 = $SHIP_4|0$
                SHIP_4_NAME = $SHIP_4_NAME|@mpp_random$
                SHIP_4_DESIGN = $SHIP_4_DESIGN|@mpp_random$
                # -- Fleet Actions -- #
                MOVE_TO_TARGET = $MOVE_TO_TARGET|@mpp_no$
                MOVE_TO_PLANET_FLAG = $MOVE_TO_PLANET_FLAG|@mpp_no$
                PATROL_PLANET_FLAG_1 = $PATROL_PLANET_FLAG_1|@mpp_no$
                PATROL_PLANET_FLAG_2 = $PATROL_PLANET_FLAG_2|@mpp_no$
                PATROL_PLANET_FLAG_3 = $PATROL_PLANET_FLAG_3|@mpp_no$
                PATROL_PLANET_FLAG_4 = $PATROL_PLANET_FLAG_4|@mpp_no$
            }

            # ---------- Limpieza de variables ----------
            clear_variable = base
        }
    }

    else = {
        mpp_log_effect_error = { EFFECT = mpp_config_fleet TEXT = "Scope no válido, se espera scope_type fleet" }
        mpp_log_error_invalid_scope = yes
    }
}

                    PATROL_PLANET_FLAG_3 = $PATROL_PLANET_FLAG_3|@mpp_no$
                    PATROL_PLANET_FLAG_4 = $PATROL_PLANET_FLAG_4|@mpp_no$
                    PATROL_PLANET_FLAG_4 = $PATROL_PLANET_FLAG_4|@mpp_no$
                }
            }
        }
    }
    else = {
        mpp_log_effect_error = { EFFECT = mpp_config_fleet TEXT = "Scope no válido, se espera scope_type fleet" }
        mpp_log_error_invalid_scope = yes
    }
}
