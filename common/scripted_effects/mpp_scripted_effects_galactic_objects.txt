mpp_go_spawn_planet = {
    optimize_memory

    mpp_middleware_position = {
        EFFECT = mpp_go_spawn_planet
        BASE_SCRIPT = galactic_objects/mpp_i_s_go_spawn_planet
        # ---------- Lógica de tipología ----------
        TYPE = $TYPE|random$ # !rev
        SIZE = $SIZE|@mpp_random$ # !rev
        # ---------- Lógica de distancia ----------
        DISTANCE = $ORBIT|@mpp_no$
        DISTANCE_OFFSET_RANGE = $ORBIT_OFFSET|@mpp_no$
        MIN_DISTANCE = $MIN_ORBIT_DISTANCE|@mpp_no$
        MAX_DISTANCE = $MAX_ORBIT_DISTANCE|@mpp_no$
        # ---------- Lógica de ángulo ----------
        ANGLE = $ANGLE|@mpp_no$
        ANGLE_OFFSET_RANGE = $ANGLE_OFFSET_RANGE|@mpp_no$
        MIN_ANGLE = $MIN_ANGLE|@mpp_no$   #value:mpp_get_min_angle_by_iterator
        MAX_ANGLE = $MAX_ANGLE|@mpp_no$ #value:mpp_get_max_angle_by_iterator
        # ---------- Lógica de loop (uso en ángulo) ----------
        ITERATION = $ITERATION|1$
        MAX_ITERATIONS = $MAX_ITERATIONS|1$
    }

    # ---------- Efectos ----------
    if = {
        limit = { exists = event_target:tPlanet }
        event_target:tPlanet = {
            [[EFFECT1] $EFFECT1$ = yes ]
            [[EFFECT2] $EFFECT2$ = yes ]
            [[EFFECT3] $EFFECT3$ = yes ]
            [[EFFECT4] $EFFECT4$ = yes ]
            [[EFFECT5] $EFFECT5$ = yes ]
            [[COMPLEX_EFFECT1] $COMPLEX_EFFECT1$ ]
            [[COMPLEX_EFFECT2] $COMPLEX_EFFECT2$ ]
            [[COMPLEX_EFFECT3] $COMPLEX_EFFECT3$ ]
            [[COMPLEX_EFFECT4] $COMPLEX_EFFECT4$ ]
            [[COMPLEX_EFFECT5] $COMPLEX_EFFECT5$ ]
        }
    }
}

mpp_go_add_asteroid_belt = {
    # Si el parámetro TYPE está definido → usar ese tipo exacto de cinturón de asteroides
    [[TYPE]
        add_asteroid_belt = {
            type = $TYPE$       # Ejemplo: rocky_asteroid_belt, icy_asteroid_belt, etc.
            radius = $RADIUS$   # Radio del cinturón
        }
    ]

    # Si no se ha definido TYPE → elegir aleatoriamente entre los tipos disponibles
    [[!TYPE]
        random_list = {
            1 = { add_asteroid_belt = { type = rocky_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = icy_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = crystal_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = debris_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = empty_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = mpp_icy_debris_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = mpp_rocky_debris_small_asteroid_belt radius = $RADIUS$ } }
        }
    ]
}

mpp_go_add_asteroids = {
    # Si se define COUNT → asignar esa cantidad a la variable temporal tCount
    [[COUNT]
        set_variable = {
            which = tCount
            value = $COUNT|1$    # $COUNT|1$ → si no se pasa COUNT, usa valor por defecto 1
        }
    ]

    # Si NO se define COUNT → generar un número aleatorio entre MIN_COUNT y MAX_COUNT
    [[!COUNT]
        set_variable_to_random_value = {
            which = tCount
            min = $MIN_COUNT|1$  # valor mínimo por defecto = 1
            max = $MAX_COUNT|1$  # valor máximo por defecto = 1 (revisar si debería ser >1)
            rounded = yes        # asegura número entero
        }
    ]

    # Bucle while: ejecuta el bloque interno tCount veces
    while = {
        count = tCount

        # Caso 1: TYPE definido → añade planeta del tipo especificado
        [[TYPE]
            mpp_go_spawn_planet = {
                TYPE = $TYPE$
                ORBIT = $ORBIT|50$
                ORBIT_OFFSET = $ORBIT_OFFSET|0$
                SIZE = $SIZE|2$
                MIN_ANGLE = $MIN_ANGLE|0$
                MAX_ANGLE = $MAX_ANGLE|360$
                EFFECT1 = mpp_setName_randomPlanet
            }
        ]

        # Caso 2: TYPE no definido → genera un asteroide aleatorio
        [[!TYPE]
            mpp_go_spawn_planet = {
                TYPE = rl_asteroids
                ORBIT = $ORBIT|50$
                ORBIT_OFFSET = $ORBIT_OFFSET|0$
                SIZE = $SIZE|2$
                MIN_ANGLE = $MIN_ANGLE|0$
                MAX_ANGLE = $MAX_ANGLE|360$
                EFFECT1 = mpp_setName_randomPlanet
            }
        ]
    }

    # Limpia la variable temporal usada en el bucle
    clear_variable = tCount
}

mpp_ao_add_random = {
    optimize_memory

    # ---------- Flags ----------
    set_variable = { which = tCategory value = $CATEGORY|@AO_CATEGORY_DEBRIS$ }

    # AO_CATEGORY_DEBRIS
    if = {
        limit = { check_variable = { which = tCategory value = @AO_CATEGORY_DEBRIS } }
        mpp_ao_add_debris = {
            LOCATION = $LOCATION|@mpp_this$
            LOCATION_TARGET = $LOCATION_TARGET|@mpp_no$
            IN_ORBIT = $IN_ORBIT|@mpp_no$
            DISTANCE = $DISTANCE|@mpp_no$
            DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|0$
            MIN_DISTANCE = $MIN_DISTANCE|1$
            MAX_DISTANCE = $MAX_DISTANCE|5$
        }
    }

    # AO_CATEGORY_BATTLEFIELD
    else_if = {
        limit = { check_variable = { which = tCategory value = @AO_CATEGORY_BATTLEFIELD } }
        mpp_ao_add_battlefield_debris = {
            LOCATION = $LOCATION|@mpp_this$
            LOCATION_TARGET = $LOCATION_TARGET|@mpp_no$
            IN_ORBIT = $IN_ORBIT|@mpp_no$
            DISTANCE = $DISTANCE|@mpp_no$
            DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|0$
            MIN_DISTANCE = $MIN_DISTANCE|1$
            MAX_DISTANCE = $MAX_DISTANCE|5$
            SHIP_DEBRIS = $SHIP_DEBRIS|0$
            SHIP_DEBRIS_MIN_DISTANCE = $SHIP_DEBRIS_MIN_DISTANCE|1$
            SHIP_DEBRIS_MAX_DISTANCE = $SHIP_DEBRIS_MAX_DISTANCE|5$
        }
    }

    # AO_CATEGORY_SHIP_DEBRIS
    else_if = {
        limit = { check_variable = { which = tCategory value = @AO_CATEGORY_SHIP_DEBRIS } }
        mpp_ao_add_ship_debris = {
            LOCATION = $LOCATION|@mpp_this$
            LOCATION_TARGET = $LOCATION_TARGET|@mpp_no$
            IN_ORBIT = $IN_ORBIT|@mpp_no$
            DISTANCE = $DISTANCE|@mpp_no$
            DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|0$
            MIN_DISTANCE = $MIN_DISTANCE|1$
            MAX_DISTANCE = $MAX_DISTANCE|10$
        }
    }
    else_if = {
        limit = { mpp_is_debug_active = yes }
        mpp_log_error = { TEXT = "mpp_add_random Categoría $CATEGORY|@AO_CATEGORY_DEBRIS$ de Ambient Objects no implementada" }
    }

    # ---------- Limpieza de variables ----------
    clear_variable = tCategory
}
