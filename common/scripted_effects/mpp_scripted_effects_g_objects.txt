mpp_go_spawn_planet = {
    inline_script = {
        script = galactic_objects/mpp_is_go_orbit_offset
        TYPE = $TYPE|random$
        ORBIT = $ORBIT|random$
        ORBIT_OFFSET = $ORBIT_OFFSET|0$
        SIZE = $SIZE|random$
        MIN_ANGLE = $MIN_ANGLE|0$
        MAX_ANGLE = $MAX_ANGLE|360$
    }

    if = {
        limit = { exists = event_target:tPlanet }
        event_target:tPlanet = {
            [[EFFECT1] planet = { $EFFECT1$ = yes } ]
            [[EFFECT2] planet = { $EFFECT2$ = yes } ]
            [[EFFECT3] planet = { $EFFECT3$ = yes } ]
            [[EFFECT4] planet = { $EFFECT4$ = yes } ]
            [[EFFECT5] planet = { $EFFECT5$ = yes } ]
        }
    }
}

mpp_go_add_asteroid_belt = {
    # Si el parámetro TYPE está definido → usar ese tipo exacto de cinturón de asteroides
    [[TYPE]
        add_asteroid_belt = {
            type = $TYPE$       # Ejemplo: rocky_asteroid_belt, icy_asteroid_belt, etc.
            radius = $RADIUS$   # Radio del cinturón
        }
    ]

    # Si no se ha definido TYPE → elegir aleatoriamente entre los tipos disponibles
    [[!TYPE]
        random_list = {
            1 = { add_asteroid_belt = { type = rocky_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = icy_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = crystal_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = debris_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = empty_asteroid_belt radius = $RADIUS$ } }
        }
    ]
}

mpp_go_add_asteroids = {
    # Si se define COUNT → asignar esa cantidad a la variable temporal tCount
    [[COUNT]
        set_variable = {
            which = tCount
            value = $COUNT|1$    # $COUNT|1$ → si no se pasa COUNT, usa valor por defecto 1
        }
    ]

    # Si NO se define COUNT → generar un número aleatorio entre MIN_COUNT y MAX_COUNT
    [[!COUNT]
        set_variable_to_random_value = {
            which = tCount
            min = $MIN_COUNT|1$  # valor mínimo por defecto = 1
            max = $MAX_COUNT|1$  # valor máximo por defecto = 1 (revisar si debería ser >1)
            rounded = yes        # asegura número entero
        }
    ]

    # Bucle while: ejecuta el bloque interno tCount veces
    while = {
        count = tCount

        # Caso 1: TYPE definido → añade planeta del tipo especificado
        [[TYPE]
            mpp_go_spawn_planet = {
                TYPE = $TYPE$
                ORBIT = $ORBIT|50$
                ORBIT_OFFSET = $ORBIT_OFFSET|0$
                SIZE = $SIZE|2$
                MIN_ANGLE = $MIN_ANGLE|0$
                MAX_ANGLE = $MAX_ANGLE|360$
                EFFECT1 = mpp_setName_randomPlanet
            }
        ]

        # Caso 2: TYPE no definido → genera un asteroide aleatorio
        [[!TYPE]
            mpp_go_spawn_planet = {
                TYPE = rl_asteroids
                ORBIT = $ORBIT|50$
                ORBIT_OFFSET = $ORBIT_OFFSET|0$
                SIZE = $SIZE|2$
                MIN_ANGLE = $MIN_ANGLE|0$
                MAX_ANGLE = $MAX_ANGLE|360$
                EFFECT1 = mpp_setName_randomPlanet
            }
        ]
    }

    # Limpia la variable temporal usada en el bucle
    clear_variable = tCount
}

mpp_ao_add = {
    optimize_memory

    # ---------- Modo creación ----------
    set_variable = { which = scriptMode value = 0 }

    # ---------- Flags ----------
    [[TYPE] set_variable = { which = hasType value = @mpp_yes } ]
    [[!TYPE] set_variable = { which = hasType value = @mpp_no } ]
    [[IN_ORBIT] set_variable = { which = inOrbit value = @mpp_yes } ]
    [[!IN_ORBIT] set_variable = { which = inOrbit value = @mpp_no } ]

    # ---------- Variables de posición ----------
    [[DISTANCE] set_variable = { which = tDistance value = $DISTANCE$ } ]
    [[!DISTANCE] set_variable = { which = tDistance value = @mpp_no } ]
    [[DISTANCE_OFFSET_RANGE] set_variable = { which = tDistanceOffsetRange value = $DISTANCE_OFFSET_RANGE$ } ]
    [[!DISTANCE_OFFSET_RANGE] set_variable = { which = tDistanceOffsetRange value = @mpp_no } ]
    [[MIN_DISTANCE] set_variable = { which = tMinDistance value = $MIN_DISTANCE$ } ]
    [[!MIN_DISTANCE] set_variable = { which = tMinDistance value = @mpp_no } ]
    [[MAX_DISTANCE] set_variable = { which = tMaxDistance value = $MAX_DISTANCE$ } ]
    [[!MAX_DISTANCE] set_variable = { which = tMaxDistance value = @mpp_no } ]

    # ---------- Lógica de posicionamiento ----------
    inline_script = {
        script = galactic_objects/mpp_is_ao_conf
    }

    # ---------- Caso 1: Tiene TYPE ----------
    if = {
        limit = { check_variable = { which = hasType value = @mpp_yes } }

        # Tiene TARGET
        [[TARGET]
            inline_script = {
                script = galactic_objects/mpp_is_ao_distance
                TYPE = $TYPE$
                TARGET = event_target:$TARGET$
            }
        ]

        # No tiene TARGET
        [[!TARGET]
            if = {
                limit = { exists = prev }
                inline_script = {
                    script = galactic_objects/mpp_is_ao_distance
                    TYPE = $TYPE$
                    TARGET = prev
                }
            }
            else_if = {
                limit = { exists = from }
                inline_script = {
                    script = galactic_objects/mpp_is_ao_distance
                    TYPE = $TYPE$
                    TARGET = from
                }
            }
        ]
    }

    # ---------- Caso 2: No tiene TYPE → usar small_debris_object ----------
    else = {

        # Tiene TARGET
        [[TARGET]
            inline_script = {
                script = galactic_objects/mpp_is_ao_distance
                TYPE = small_debris_object
                TARGET = event_target:$TARGET$
            }
        ]

        # No tiene TARGET
        [[!TARGET]
            if = {
                limit = { exists = prev }
                inline_script = {
                    script = galactic_objects/mpp_is_ao_distance
                    TYPE = small_debris_object
                    TARGET = prev
                }
            }
            else_if = {
                limit = { exists = from }
                inline_script = {
                    script = galactic_objects/mpp_is_ao_distance
                    TYPE = small_debris_object
                    TARGET = from
                }
            }
        ]
    }

    # ---------- Limpieza de variables ----------
    clear_variable = scriptMode
    clear_variable = hasType
    clear_variable = inOrbit
    clear_variable = tDistance
    clear_variable = tDistanceOffsetRange
    clear_variable = tMinDistance
    clear_variable = tMaxDistance
}

                TYPE = broken_star_lifting_platform
                TARGET = tStar
                DISTANCE = $DISTANCE|@mpp_no$
                DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|0$
                MIN_DISTANCE = $MIN_DISTANCE|@defaultDebrisMinDistance$
                MAX_DISTANCE = $MAX_DISTANCE|@defaultDebrisMaxDistance$
            }

            last_created_ambient_object = {
                #noinspection ParadoxScriptUnusedDynamicValue
                save_event_target_as = tPlatform
            }

            [[ADD_EVENT]
                # System
                prev = {
                    #noinspection ParadoxScriptUnusedDynamicValue
                    set_star_flag = star_lifting_system
                }

                event_target:broken_star_lifting_platform = {
                    #noinspection ParadoxScriptUnusedDynamicValue
                    save_global_event_target_as = mpp_broken_star_lifting_platform
                    save_global_event_target_as = broken_star_lifting_platform

                    add_anomaly = {
                        category = star_lifting_category
                    }
                }
            ]

            [[ADD_BATTLEFIELD]
                mpp_ao_add_battlefield_debris = {
                    TARGET = tPlatform
                    MIN_DISTANCE = $BATTLEFIELD_MIN_DISTANCE|1$
                    MAX_DISTANCE = $BATTLEFIELD_MAX_DISTANCE|5$
                }
            ]

            [[SMALL_DEBRIS]
                mpp_ao_add_debris = {
                    TYPE = small_debris_object
                    TARGET = tPlatform
                    COUNT = $SMALL_DEBRIS$
                    MIN_DISTANCE = $DEBRIS_MIN_DISTANCE|1$
                    MAX_DISTANCE = $DEBRIS_MAX_DISTANCE|5$
                }
            ]
        }
    }
