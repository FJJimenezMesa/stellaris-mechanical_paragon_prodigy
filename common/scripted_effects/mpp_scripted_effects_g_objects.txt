mpp_go_spawn_planet = {
    inline_script = {
        script = galactic_objects/mpp_is_go_orbit_offset
        TYPE = $TYPE|random$
        ORBIT = $ORBIT|random$
        ORBIT_OFFSET = $ORBIT_OFFSET|0$
        SIZE = $SIZE|random$
        MIN_ANGLE = $MIN_ANGLE|0$
        MAX_ANGLE = $MAX_ANGLE|360$
    }

    if = {
        limit = { exists = event_target:tPlanet }
        event_target:tPlanet = {
            [[EFFECT1] planet = { $EFFECT1$ = yes } ]
            [[EFFECT2] planet = { $EFFECT2$ = yes } ]
            [[EFFECT3] planet = { $EFFECT3$ = yes } ]
            [[EFFECT4] planet = { $EFFECT4$ = yes } ]
            [[EFFECT5] planet = { $EFFECT5$ = yes } ]
        }
    }
}

mpp_go_add_asteroid_belt = {
    # Si el parámetro TYPE está definido → usar ese tipo exacto de cinturón de asteroides
    [[TYPE]
        add_asteroid_belt = {
            type = $TYPE$       # Ejemplo: rocky_asteroid_belt, icy_asteroid_belt, etc.
            radius = $RADIUS$   # Radio del cinturón
        }
    ]

    # Si no se ha definido TYPE → elegir aleatoriamente entre los tipos disponibles
    [[!TYPE]
        random_list = {
            1 = { add_asteroid_belt = { type = rocky_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = icy_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = crystal_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = debris_asteroid_belt radius = $RADIUS$ } }
            1 = { add_asteroid_belt = { type = empty_asteroid_belt radius = $RADIUS$ } }
        }
    ]
}

mpp_go_add_asteroids = {
    # Si se define COUNT → asignar esa cantidad a la variable temporal tCount
    [[COUNT]
        set_variable = {
            which = tCount
            value = $COUNT|1$    # $COUNT|1$ → si no se pasa COUNT, usa valor por defecto 1
        }
    ]

    # Si NO se define COUNT → generar un número aleatorio entre MIN_COUNT y MAX_COUNT
    [[!COUNT]
        set_variable_to_random_value = {
            which = tCount
            min = $MIN_COUNT|1$  # valor mínimo por defecto = 1
            max = $MAX_COUNT|1$  # valor máximo por defecto = 1 (revisar si debería ser >1)
            rounded = yes        # asegura número entero
        }
    ]

    # Bucle while: ejecuta el bloque interno tCount veces
    while = {
        count = tCount

        # Caso 1: TYPE definido → añade planeta del tipo especificado
        [[TYPE]
            mpp_go_spawn_planet = {
                TYPE = $TYPE$
                ORBIT = $ORBIT|50$
                ORBIT_OFFSET = $ORBIT_OFFSET|0$
                SIZE = $SIZE|2$
                MIN_ANGLE = $MIN_ANGLE|0$
                MAX_ANGLE = $MAX_ANGLE|360$
                EFFECT1 = mpp_setName_randomPlanet
            }
        ]

        # Caso 2: TYPE no definido → genera un asteroide aleatorio
        [[!TYPE]
            mpp_go_spawn_planet = {
                TYPE = rl_asteroids
                ORBIT = $ORBIT|50$
                ORBIT_OFFSET = $ORBIT_OFFSET|0$
                SIZE = $SIZE|2$
                MIN_ANGLE = $MIN_ANGLE|0$
                MAX_ANGLE = $MAX_ANGLE|360$
                EFFECT1 = mpp_setName_randomPlanet
            }
        ]
    }

    # Limpia la variable temporal usada en el bucle
    clear_variable = tCount
}

mpp_ao_add = {
    optimize_memory

    # ---------- Flags ----------
    [[TYPE] set_variable = { which = hasType value = @mpp_yes } ]
    [[!TYPE] set_variable = { which = hasType value = @mpp_no } ]
    [[TARGET] set_variable = { which = hasTarget value = @mpp_yes } ]
    [[!TARGET] set_variable = { which = hasTarget value = @mpp_no } ]
    set_variable = { which = inOrbit value = $IN_ORBIT|@mpp_no$ }
    log_error = "InOrbit \\[this.inOrbit]"

    # ---------- Caso 1: Tiene TYPE ----------
    if = {
        limit = { check_variable = { which = hasType value = @mpp_yes } }

        # Tiene TARGET
        if = {
            limit = { check_variable = { which = hasTarget value = @mpp_yes } }
            # IN_ORBIT activo
            if = {
                limit = { check_variable = { which = inOrbit value = @mpp_yes } }
                log_error = "Crea $TYPE$ con target inOrbit"
                create_ambient_object = {
                    type = $TYPE$
                    location = event_target:$TARGET|prev$           # No usará el default
                    use_3d_location = yes
                    entity_offset_height = @inOrbitOffsetHeight
                }
            }
            # Solo TARGET
            else = {
                log_error = "Crea $TYPE$ con target"
                create_ambient_object = {
                    type = $TYPE$
                    location = event_target:$TARGET|prev$           # No usará el default
                }
            }
        }
        # No tiene TARGET
        else = {
            # Solo IN_ORBIT
            if = {
                limit = { check_variable = { which = inOrbit value = @mpp_yes } }
                log_error = "Crea $TYPE$ sin target inOrbit"
                create_ambient_object = {
                    type = $TYPE$
                    location = prev
                    use_3d_location = yes
                    entity_offset_height = @inOrbitOffsetHeight
                }
            }
            # Caso base
            else = {
                log_error = "Crea $TYPE$ sin target"
                create_ambient_object = {
                    type = $TYPE$
                    location = prev
                }
            }
        }
    }

    # ---------- Caso 2: No tiene TYPE → usar small_debris_object ----------
    else = {
        # Tiene TARGET
        if = {
            limit = { check_variable = { which = hasTarget value = @mpp_yes } }
            # IN_ORBIT activo
            if = {
                limit = { check_variable = { which = inOrbit value = @mpp_yes } }
                log_error = "Crea small_debris_object con target inOrbit"
                create_ambient_object = {
                    type = small_debris_object
                    location = event_target:$TARGET|prev$           # No usará el default
                    use_3d_location = yes
                    entity_offset_height = @inOrbitOffsetHeight
                }
            }
            # Solo TARGET
            else = {
                log_error = "Crea small_debris_object con target"
                create_ambient_object = {
                    type = small_debris_object
                    location = event_target:$TARGET|prev$           # No usará el default
                }
            }
        }
        # No TARGET
        else = {
            # Solo IN_ORBIT
            if = {
                limit = { check_variable = { which = inOrbit value = @mpp_yes } }
                log_error = "Crea small_debris_object sin target inOrbit"
                create_ambient_object = {
                    type = small_debris_object
                    location = prev
                    use_3d_location = yes
                    entity_offset_height = @inOrbitOffsetHeight
                }
            }
            # Caso base
            else = {
                log_error = "Crea small_debris_object sin target"
                create_ambient_object = {
                    type = small_debris_object
                    location = prev
                }
            }
        }
    }

    last_created_ambient_object = {
        [[TARGET]
            mpp_ao_set_location = {
                TARGET = $TARGET$
                DISTANCE = $DISTANCE|@mpp_no$
                DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|0$
                MIN_DISTANCE = $MIN_DISTANCE|@mpp_no$
                MAX_DISTANCE = $MAX_DISTANCE|@mpp_no$
            }
        ]

        [[!TARGET]
            mpp_ao_set_location = {
                DISTANCE = $DISTANCE|@mpp_no$
                DISTANCE_OFFSET_RANGE = $DISTANCE_OFFSET_RANGE|0$
                MIN_DISTANCE = $MIN_DISTANCE|@mpp_no$
                MAX_DISTANCE = $MAX_DISTANCE|@mpp_no$
            }
        ]
    }

    # ---------- Limpieza de variables ----------
    clear_variable = hasType
    clear_variable = hasTarget
    clear_variable = inOrbit
}

mpp_ao_set_location = {
    optimize_memory

    # ---------- Flags ----------
    set_variable = { which = tDistance value = $DISTANCE|@mpp_no$ }
    set_variable = { which = tDistanceOffsetRange value = $DISTANCE_OFFSET_RANGE|@mpp_no$ }
    set_variable = { which = tMinDistance value = $MIN_DISTANCE|@mpp_no$ }
    set_variable = { which = tMaxDistance value = $MAX_DISTANCE|@mpp_no$ }

    # ---------- Caso 1: DISTANCE presente ----------
    if = {
        limit = { NOT = { check_variable = { which = tDistance value = @mpp_no } } }

        # Offset definido
        if = {
            limit = { NOT = { check_variable = { which = tDistanceOffsetRange value = @mpp_no } } }
            set_variable = { which = tOffsetMax value = tDistanceOffsetRange }
            if = {
                limit = { check_variable = { which = tOffsetMax value < -50 } }
                set_variable = { which = tOffsetMax value = -50 }
            }
            if = {
                limit = { check_variable = { which = tOffsetMax value > 49 } }
                set_variable = { which = tOffsetMax value = 49 }
            }
        }
        else = {
            set_variable = { which = tOffsetMax value = 0 }
        }

        set_variable = { which = tOffsetMin value = tOffsetMax }

        # Convertir min a negativo si no es 0 menor que 0
        if = {
            limit = { NOT = { check_variable = { which = tOffsetMin value <= 0 } } }
            multiply_variable = { which = tOffsetMin value = -1 }
        }

        # Valor random de offset
        set_variable_to_random_value = { which = tOffset MIN = tOffsetMin MAX = tOffsetMax rounded = yes }

        # Determinar ubicación con o sin target
        [[TARGET]
            log_error = "distance con target"
            inline_script = {
                script = galactic_objects/mpp_is_ao_set_location_offset
                TARGET = event_target:$TARGET|prev$     # No usará el default
                DISTANCE = $DISTANCE|0$                 # No usará el default
            }
        ]

        [[!TARGET]
            log_error = "distance no target y offset"
            inline_script = {
                script = galactic_objects/mpp_is_ao_set_location_offset
                TARGET = prev
                DISTANCE = $DISTANCE|0$                 # No usará el default
            }
        ]

        # ---------- Limpieza de variables ----------
        clear_variable = tOrbitOffset
        clear_variable = tOffsetMin
        clear_variable = tOffsetMax
    }

    # ---------- Caso 2: DISTANCE no presente → usar MIN/MAX ----------
    else = {
        # MIN_DISTANCE
        if = {
            limit = { NOT = { check_variable = { which = tMinDistance value = @mpp_no } } }
            if = {
                limit = { check_variable_arithmetic = { which = tMinDistance value < -50 } }
                set_variable = { which = tMinDistance value = -50 }
            }
        }
        else = {
            set_variable = { which = tMinDistance value = 1 }
        }

        # MAX_DISTANCE
        if = {
            limit = { NOT = { check_variable = { which = tMaxDistance value = @mpp_no } } }
            if = {
                limit = { check_variable_arithmetic = { which = tMaxDistance value > 299 } }
                set_variable = { which = tMaxDistance value = 299 }
            }
        }
        else = {
            set_variable = { which = tMaxDistance value = 1 }
        }

        # Valor random de distancia
        set_variable_to_random_value = { which = tDistance min = tMinDistance max = tMaxDistance rounded = yes }

        log_error = "MIN \\[this.tMinDistance]"
        log_error = "MAX \\[this.tMaxDistance]"
        log_error = "DISTANCE \\[this.tDistance]"

        # Ubicación con target
        if = {
            limit = { check_variable = { which = hasTarget value = @mpp_yes } }
            log_error = "location con target"
            inline_script = {
                script = galactic_objects/mpp_is_ao_set_location
                TARGET = event_target:$TARGET$
            }
        }
        # Ubicación sin target
        else = {
            log_error = "location sin target"
            inline_script = {
                script = galactic_objects/mpp_is_ao_set_location
                TARGET = this
            }
        }
    }

    # ---------- Limpieza de variables ----------
    clear_variable = tDistance
    clear_variable = tDistanceOffsetRange
    clear_variable = tMinDistance
    clear_variable = tMaxDistance
}
