mpp_setName = {
    [[RANDOM]
        random_list = {
            1 = { mpp_setName = { RANDOM_PROMETHEUS = yes } }
            1 = { mpp_setName = { RANDOM_PROMETHEUS_GODS = yes } }
            1 = { mpp_setName = { RANDOM_DARK = yes } }
            1 = { mpp_setName = { RANDOM_FIRE = yes } }
            1 = { mpp_setName = { RANDOM_COLD = yes } }
            1 = { mpp_setName = { RANDOM_DEAD = yes } }
            1 = { mpp_setName = { RANDOM_TOXIC = yes } }
            1 = { mpp_setName = { RANDOM_NANO = yes } }
        }
    ]

    [[RANDOM_PROMETHEUS]
        set_name = {
            key = "MPP_NAME_randomPrometheus"
            variable_string = "\\[This.MPPGetNameRandomPrometheus\\]"
        }
    ]

    [[RANDOM_PROMETHEUS_GODS]
        set_name = {
            key = "MPP_NAME_randomPrometheusGods"
            variable_string = "\\[This.MPPGetNameRandomPrometheusGods\\]"
        }
    ]

    [[RANDOM_DARK]
        set_name = {
            key = "MPP_NAME_randomDark"
            variable_string = "\\[This.MPPGetNameRandomDark\\]"
        }
    ]

    [[RANDOM_FIRE]
        set_name = {
            key = "MPP_NAME_randomFire"
            variable_string = "\\[This.MPPGetNameRandomFire\\]"
        }
    ]

    [[RANDOM_COLD]

        set_name = {
            key = "MPP_NAME_randomCold"
            variable_string = "\\[This.MPPGetNameRandomCold\\]"
        }
    ]

    [[RANDOM_DEAD]
        set_name = {
            key = "MPP_NAME_randomDead"
            variable_string = "\\[This.MPPGetNameRandomDead\\]"
        }
    ]

    [[RANDOM_TOXIC]
        set_name = {
            key = "MPP_NAME_randomToxic"
            variable_string = "\\[This.MPPGetNameRandomToxic\\]"
        }
    ]

    [[RANDOM_NANO]
        set_name = {
            key = "MPP_NAME_randomNano"
            variable_string = "\\[This.MPPGetNameRandomNano\\]"
        }
    ]

    [[RANDOM_TERM]
        set_name = {
            key = "MPP_NAME_randomTerm"
            variable_string = "\\[This.MPPGetNameRandomTerm\\]"
        }
    ]

    [[PC_TYPE]
        if = {
            limit = { mpp_is_planet_scope = yes }

            if = {
                limit = { mpp_is_planet_class_asteroid = yes }
                set_name = MPP_NAME_ASTEROID
            }
            else_if = {
                limit = { mpp_is_planet_class_star = yes }
                set_name = MPP_NAME_STAR
            }
            else_if = {
                limit = { mpp_is_planet_class_habitat = yes }
                set_name = MPP_NAME_HABITAT
            }
            else_if = {
                limit = { mpp_is_planet_class_ring = yes }
                set_name = MPP_NAME_RING
            }
            else_if = {
                limit = { mpp_is_planet_class_planet = yes }
                if = {
                    limit = { is_moon = no }
                    set_name = MPP_NAME_PLANET
                }
                else = {
                    set_name = MPP_NAME_MOON
                }
            }
            else = {
                set_name = MPP_NAME_GALACIC_OBJECT
            }
        }
    ]

    [[STAR_NAME]
        if = {
            limit = { mpp_is_planet_scope = yes }

            save_event_target_as = tPlanet

            solar_system = {
                star = {
                    event_target:tPlanet = {
                        set_name = {
                            key = "MPP_NAME_starName"
                            variable_string = "\\[Prev.GetName\\]"
                        }
                    }
                }
            }
        }
    ]
}

mpp_changeName_add = {

    [[SPACE]
        set_name = {
            key = "MPP_NAME_add_space"
            variable_string = "\\[This.GetName\\]"
        }
    ]

    [[SLASH]
        set_name = {
            key = "MPP_NAME_add_slash"
            variable_string = "\\[This.GetName\\]"
        }
    ]

    [[STAR_NAME]
        if = {
            limit = { mpp_is_planet_scope = yes }

            save_event_target_as = tPlanet

            solar_system = {
                star = {
                    event_target:tPlanet = {
                        set_name = {
                            key = "MPP_NAME_add_starName"
                            variable_string = "\\[This.GetName\\]"
                            variable_string = "\\[Prev.GetName\\]"
                        }
                    }
                }
            }
        }
    ]

    [[RAND_TERM]
        set_name = {
            key = "MPP_NAME_add_randomTerm"
            variable_string = "\\[This.GetName\\]"
            variable_string = "\\[This.MPPGetNameRandomTerm\\]"
        }
    ]

    [[PC_TYPE]
        if = {
            limit = { mpp_is_planet_scope = yes }

            set_name = {
                key = "MPP_NAME_add_pcType"
                variable_string = "\\[This.GetName\\]"
                variable_string = "\\[This.MPPGetNamePcType\\]"
            }
        }
    ]

    [[NUMBER]
        set_variable = {
            which = tNumber
            value = $NUMBER|0$
        }

        # NORMALIZE
        set_variable = {
            which = tNormalize
            value = $NORMALIZE|0$
        }
        if = {
            limit = {
                check_variable_arithmetic = { which = tNormalize value >= 2 }
                check_variable_arithmetic = { which = tNumber value < 10 }
            }
            set_name = {
                key = "MPP_NAME_add_0"
                variable_string = "\\[This.GetName\\]"
            }
        }
        if = {
            limit = {
                check_variable_arithmetic = { which = tNormalize value >= 3 }
                check_variable_arithmetic = { which = tNumber value < 100 }
            }
            set_name = {
                key = "MPP_NAME_add_0"
                variable_string = "\\[This.GetName\\]"
            }
        }

        set_name = {
            key = "MPP_NAME_add_number"
            variable_string = "\\[This.GetName\\]"
            variable_string = "\\[This.tNumber\\]"
        }
        clear_variable = tNumber
    ]

    [[PC_TYPE_COUNT]
        if = {
            limit = { mpp_is_planet_scope = yes }

            set_variable = {
                which = pcTypeCount
                value = 0
            }
            save_event_target_as = rootPlanet

            from = {
                # System scope
                if = {
                    limit = { mpp_is_galactic_object = yes }

                    every_system_planet = {
                        # Planet scope
                        if = {
                            limit = {
                                OR = {
                                    # --- ASTEROIDS ---
                                    AND = {
                                        # Planet scope
                                        is_planet_class = pc_asteroid
                                        event_target:rootPlanet = {
                                            # Root planet scope
                                            is_planet_class = pc_asteroid
                                        }
                                    }
                                    AND = { is_planet_class = pc_ice_asteroid event_target:rootPlanet = { is_planet_class = pc_ice_asteroid } }
                                    AND = { is_planet_class = pc_crystal_asteroid event_target:rootPlanet = { is_planet_class = pc_crystal_asteroid } }

                                    AND = { is_planet_class = pc_m_star event_target:rootPlanet = { is_planet_class = pc_m_star } }
                                    AND = { is_planet_class = pc_m_giant_star event_target:rootPlanet = { is_planet_class = pc_m_giant_star } }
                                    AND = { is_planet_class = pc_t_star event_target:rootPlanet = { is_planet_class = pc_t_star } }
                                    AND = { is_planet_class = pc_toxoid_star event_target:rootPlanet = { is_planet_class = pc_toxoid_star } }
                                    AND = { is_planet_class = pc_neutron_star event_target:rootPlanet = { is_planet_class = pc_neutron_star } }
                                    AND = { is_planet_class = pc_pulsar event_target:rootPlanet = { is_planet_class = pc_pulsar } }

                                    # --- HABITATS ---
                                    AND = { is_planet_class = pc_habitat event_target:rootPlanet = { is_planet_class = pc_habitat } }

                                    AND = { is_planet_class = pc_crystal_habitat event_target:rootPlanet = { is_planet_class = pc_crystal_habitat } }

                                    # --- RINGWORLDS ---
                                    AND = { is_planet_class = pc_cybrex event_target:rootPlanet = { is_planet_class = pc_cybrex } }
                                    AND = { is_planet_class = pc_ringworld_habitable event_target:rootPlanet = { is_planet_class = pc_ringworld_habitable } }
                                    AND = { is_planet_class = pc_ringworld_habitable_damaged event_target:rootPlanet = { is_planet_class = pc_ringworld_habitable_damaged } }
                                    AND = { is_planet_class = pc_ringworld_seam event_target:rootPlanet = { is_planet_class = pc_ringworld_seam } }
                                    AND = { is_planet_class = pc_ringworld_seam_damaged event_target:rootPlanet = { is_planet_class = pc_ringworld_seam_damaged } }
                                    AND = { is_planet_class = pc_ringworld_shielded event_target:rootPlanet = { is_planet_class = pc_ringworld_shielded } }

                                    AND = { is_planet_class = pc_ringworld_tech_damaged event_target:rootPlanet = { is_planet_class = pc_ringworld_tech_damaged } }
                                    AND = { is_planet_class = pc_shattered_ring_habitable event_target:rootPlanet = { is_planet_class = pc_shattered_ring_habitable } }

                                    # --- PLANETS ---
                                    AND = { is_planet_class = pc_alpine event_target:rootPlanet = { is_planet_class = pc_alpine } }
                                    AND = { is_planet_class = pc_arctic event_target:rootPlanet = { is_planet_class = pc_arctic } }
                                    AND = { is_planet_class = pc_arid event_target:rootPlanet = { is_planet_class = pc_arid } }

                                    AND = { is_planet_class = pc_city event_target:rootPlanet = { is_planet_class = pc_city } }
                                    AND = { is_planet_class = pc_continental event_target:rootPlanet = { is_planet_class = pc_continental } }
                                    AND = { is_planet_class = pc_desert event_target:rootPlanet = { is_planet_class = pc_desert } }
                                    AND = { is_planet_class = pc_egg_cracked event_target:rootPlanet = { is_planet_class = pc_egg_cracked } }
                                    AND = { is_planet_class = pc_frozen event_target:rootPlanet = { is_planet_class = pc_frozen } }
                                    AND = { is_planet_class = pc_gaia event_target:rootPlanet = { is_planet_class = pc_gaia } }

                                    AND = { is_planet_class = pc_machine_broken event_target:rootPlanet = { is_planet_class = pc_machine_broken } }

                                    AND = { is_planet_class = pc_nuked event_target:rootPlanet = { is_planet_class = pc_nuked } }

                which = tNormalize
                value = $NORMALIZE|0$

                limit = {
                    check_variable_arithmetic = { which = tNormalize value >= 2 }

mpp_setName_randomPlanet = {
    if = {
        limit = { mpp_is_planet_class_asteroid = yes }

        mpp_setName = { STAR_NAME = yes }
        mpp_changeName_add = { SLASH = yes }
        mpp_changeName_add = { PC_TYPE = yes }
        if = {
            limit = { is_variable_set = pOrbit }
            mpp_changeName_add = { NUMBER = pOrbit NORMALIZE = 3 }
        }
        mpp_changeName_add = { SLASH = yes }
        mpp_changeName_add = { PC_TYPE_COUNT = yes NORMALIZE = 2 }
        mpp_changeName_add = { RAND_TERM = yes }

    } else_if = {
        limit = { mpp_is_planet_scope = yes }

        mpp_setName = { PC_TYPE = yes }
        mpp_changeName_add = { SPACE = yes }
        mpp_changeName_add = { RAND_TERM = yes }
    }
}
