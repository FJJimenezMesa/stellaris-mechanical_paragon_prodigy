# Guardamos valores recibidos en variables locales
set_variable = { which = inOrbit value = $IN_ORBIT$ }
set_variable = { which = tAngle value = $LOCATION_ANGLE$ }

# ============================================================
# CASO 1: El objeto debe colocarse "en órbita" (in_system)
# ============================================================
if = {
    limit = { check_variable = { which = inOrbit value = @mpp_yes } }

    # ------------------------------------------------------------
    # Subcaso 1A: Ángulo definido explícitamente
    # ------------------------------------------------------------
    if = {
        limit = {
            NOT = {
                check_variable = { which = tAngle value = @mpp_no }
                check_variable = { which = tAngle value = @mpp_random }
            }
        }
        set_location = {
            $T_TARGET$                          # Target del set_location
            distance = $LOCATION_DISTANCE$      # Distancia orbital
            angle = $LOCATION_ANGLE$            # Ángulo definido
            direction = in_system               # Dirección orbital
        }
        # ---------- Debug ----------
        mpp_log_script_effect = {
            EFFECT = $EFFECT$ SCRIPT = mpp_i_s_set_location
            TEXT = "set_location en $LOCATION$ a distancia $LOCATION_DISTANCE$ con ángulo $LOCATION_ANGLE$ (in_system)"
        }
    }

    # ------------------------------------------------------------
    # Subcaso 1B: Ángulo aleatorio
    # ------------------------------------------------------------
    else = {
        set_location = {
            $T_TARGET$
            distance = $LOCATION_DISTANCE$
            angle = random
            direction = in_system
        }
        # ---------- Debug ----------
        mpp_log_script_effect = {
            EFFECT = $EFFECT$ SCRIPT = mpp_i_s_set_location
            TEXT = "set_location en $LOCATION$ a distancia $LOCATION_DISTANCE$ con ángulo random (in_system)"
        }
    }
}

# ============================================================
# CASO 2: El objeto NO está en órbita (posición libre, out_system)
# ============================================================
else = {
    # ------------------------------------------------------------
    # Subcaso 2A: Ángulo definido explícitamente
    # ------------------------------------------------------------
    if = {
        limit = {
            NOT = {
                check_variable = { which = tAngle value = @mpp_no }
                check_variable = { which = tAngle value = @mpp_random }
            }
        }
        set_location = {
            $T_TARGET$
            distance = $LOCATION_DISTANCE$
            angle = $LOCATION_ANGLE$
            direction = out_system
        }
        # ---------- Debug ----------
        mpp_log_script_effect = {
            EFFECT = $EFFECT$ SCRIPT = mpp_i_s_set_location
            TEXT = "set_location en $T_TARGET$ a distancia $LOCATION_DISTANCE$ con ángulo $LOCATION_ANGLE$ (out_system)"
        }
    }

    # ------------------------------------------------------------
    # Subcaso 2B: Ángulo aleatorio
    # ------------------------------------------------------------
    else = {
        set_location = {
            $T_TARGET$
            distance = $LOCATION_DISTANCE$
            angle = random
            direction = out_system
        }
        # ---------- Debug ----------
        mpp_log_script_effect = {
            EFFECT = $EFFECT$ SCRIPT = mpp_i_s_set_location
            TEXT = "set_location en $T_TARGET$ a distancia $LOCATION_DISTANCE$ con ángulo random (out_system)"
        }
    }
}

# ---------- Limpieza de variables ----------
clear_variable = inOrbit
clear_variable = tAngle
