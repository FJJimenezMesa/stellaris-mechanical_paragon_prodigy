limit = { check_variable_arithmetic = { which = tDistance value = $F_DISTANCE$ } }

# ---------- Caso 1: MODO CREACIÓN ----------
if = {
    limit = { check_variable = { which = scriptMode value = 0 } }

    # IN ORBIT
    if = {
        limit = { check_variable = { which = inOrbit value = @mpp_yes } }
        create_ambient_object = {
            type = $TYPE$
            location = $TARGET$
            entity_offset = $F_DISTANCE$
            entity_offset_height = -10
            entity_offset_angle = { min = 0 max = 359 }
        }
    }
    else = {
        create_ambient_object = {
            type = $TYPE$
            location = $TARGET$
            entity_offset = $F_DISTANCE$
            entity_offset_angle = { min = 0 max = 359 }
        }
    }

    if = {
        limit = { mpp_is_debug_active = yes }
        log_error = "🛰️ - MPP - effect:mpp_ao_add $TYPE$ en $TARGET$ a $F_DISTANCE$ inOrbt \\[this.inOrbit]"
    }
}

# ---------- Caso 2: MODO LOCATION ----------
else = {
    # IN ORBIT
    if = {
        limit = { check_variable = { which = inOrbit value = @mpp_yes } }
        set_location = {
            target = $TARGET$
            distance = $F_DISTANCE$
            angle = random
            direction = in_system
        }
    }
    else = {
        set_location = {
            target = $TARGET$
            distance = $F_DISTANCE$
            angle = random
            direction = out_system
        }
    }

    if = {
        limit = { mpp_is_debug_active = yes }
        log_error = "🛰️ - MPP - effect:mpp_ao_set_location: \\[this.tTarget] en $TARGET$ a $F_DISTANCE$"
    }
}
