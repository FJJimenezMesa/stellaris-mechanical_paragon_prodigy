set_variable = { which = inOrbit value = $IN_ORBIT$ }

if = {
    limit = { exists = event_target:$LOCATION_TARGET$ }
    # IN ORBIT
    if = {
        limit = { check_variable = { which = inOrbit value = @mpp_yes } }
        create_ambient_object = {
            type = $TYPE$
            location = event_target:$LOCATION_TARGET$
            entity_offset = $LOCATION_DISTANCE$
            entity_offset_height = -10

            # TODO añadir cálculo de angle
            entity_offset_angle = { min = 0 max = 359 }
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en event_target:$LOCATION_TARGET$ a $LOCATION_DISTANCE$ inOrbt"
        }
    }
    else = {
        create_ambient_object = {
            type = $TYPE$
            location = event_target:$LOCATION_TARGET$
            entity_offset = $LOCATION_DISTANCE$

            # TODO añadir cálculo de angle
            entity_offset_angle = { min = 0 max = 359 }
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en event_target:$LOCATION_TARGET$ a $LOCATION_DISTANCE$"
        }
    }
}
else_if = {
    limit = { exists = prev.event_target:$LOCATION_TARGET$ }

    # IN ORBIT
    if = {
        limit = { check_variable = { which = inOrbit value = @mpp_yes } }
        create_ambient_object = {
            type = $TYPE$
            location = prev.event_target:$LOCATION_TARGET$
            entity_offset = $LOCATION_DISTANCE$
            entity_offset_height = -10

            # TODO añadir cálculo de angle
            entity_offset_angle = { min = 0 max = 359 }
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en prev.event_target:$LOCATION_TARGET$ a $LOCATION_DISTANCE$ inOrbt"
        }
    }
    else = {
        create_ambient_object = {
            type = $TYPE$
            location = prev.event_target:$LOCATION_TARGET$
            entity_offset = $LOCATION_DISTANCE$

            # TODO añadir cálculo de angle
            entity_offset_angle = { min = 0 max = 359 }
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en prev.event_target:$LOCATION_TARGET$ a $LOCATION_DISTANCE$"
        }
    }
}
else = {
    mpp_log_script_error = {
        EFFECT = $EFFECT$ SCRIPT = mpp_i_s_ao_add_with_location_target
        TEXT = "LOCATION_TARGET $LOCATION_TARGET$ y prev.$LOCATION_TARGET$ no válidos"
    }
}
