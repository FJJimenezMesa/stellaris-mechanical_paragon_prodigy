set_variable_to_random_value = {
    which = tDistance
    min = $MIN_DISTANCE$
    max = $MAX_DISTANCE$
    rounded = yes
}

if = {
    limit = { check_variable_arithmetic = { which = tDistance value >= 0 } }
    if = {
        limit = { check_variable_arithmetic = { which = tDistance value < 10 } }
        inline_script = {
            script = "galactic_objects/mpp_i_s_g_o_set_location_0_10" B_DISTANCE = 0
        }
    }
    else_if = {
        limit = { check_variable_arithmetic = { which = tDistance value < 20 } }
        inline_script = {
            script = "galactic_objects/mpp_i_s_g_o_set_location_0_10" B_DISTANCE = 10
        }
    }
    else_if = {
        limit = { check_variable_arithmetic = { which = tDistance value < 30 } }
        inline_script = {
            script = "galactic_objects/mpp_i_s_g_o_set_location_0_10" B_DISTANCE = 20
        }
    }
    else_if = {
        limit = { check_variable_arithmetic = { which = tDistance value < 40 } }
        inline_script = {
            script = "galactic_objects/mpp_i_s_g_o_set_location_0_10" B_DISTANCE = 30
        }
    }
    else_if = {
        limit = { check_variable_arithmetic = { which = tDistance value < 50 } }
        inline_script = {
            script = "galactic_objects/mpp_i_s_g_o_set_location_0_10" B_DISTANCE = 40
        }
    }
}

    if = {
        limit = { check_variable_arithmetic = { which = tDistance value >= -10 } }

        limit = { check_variable_arithmetic = { which = tDistance value >= -30 } }
        inline_script = {
            script = "galactic_objects/mpp_i_s_g_o_set_location_0_10" B_DISTANCE = -30
        }

    else_if = {
        limit = { check_variable_arithmetic = { which = tDistance value >= -40 } }
        inline_script = {
            script = "galactic_objects/mpp_i_s_g_o_set_location_0_10" B_DISTANCE = -40

        limit = { check_variable_arithmetic = { which = tDistance value >= -50 } }
