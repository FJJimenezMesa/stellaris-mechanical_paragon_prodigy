limit = { check_variable_arithmetic = { which = tDistance value = $F_DISTANCE$ } }

# ---------- Caso 1: MODO CREACIÓN ----------
if = {
    limit = { check_variable = { which = scriptMode value = 0 } }

    # IN ORBIT
    if = {
        limit = { check_variable = { which = inOrbit value = @mpp_yes } }
        create_ambient_object = {
            type = $TYPE$
            location = $TARGET$
            entity_offset = $F_DISTANCE$
            entity_offset_height = -10
            entity_offset_angle = { min = 0 max = 359 }
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en $TARGET$ a $F_DISTANCE$ inOrbt"
        }
    }
    else = {
        create_ambient_object = {
            type = $TYPE$
            location = $TARGET$
            entity_offset = $F_DISTANCE$
            entity_offset_angle = { min = 0 max = 359 }
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en $TARGET$ a $F_DISTANCE$"
        }
    }
}

# ---------- Caso 2: MODO LOCATION ----------
else = {
    # IN ORBIT
    if = {
        limit = { check_variable = { which = inOrbit value = @mpp_yes } }
        set_location = {
            target = $TARGET$
            distance = $F_DISTANCE$
            angle = random
            direction = in_system
        }
        # ---------- Debug ----------
        mpp_log_script_config = {
            EFFECT = $EFFECT$ SCRIPT = mpp_i_s_o TEXT = "set_location en $TARGET$ a $F_DISTANCE$"
        }
    }
    else = {
        set_location = {
            target = $TARGET$
            distance = $F_DISTANCE$
            angle = random
            direction = out_system
        }
        # ---------- Debug ----------
        mpp_log_script_config = {
            EFFECT = $EFFECT$ SCRIPT = mpp_i_s_o TEXT = "set_location en $TARGET$ a $F_DISTANCE$"
        }
    }
}
