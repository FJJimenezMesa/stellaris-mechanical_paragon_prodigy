# Guardamos valores recibidos en variables locales
set_variable = { which = inOrbit value = $IN_ORBIT$ }
set_variable = { which = tAngle value = $LOCATION_ANGLE$ }

# ============================================================
# CASO 1: El objeto debe colocarse "en órbita"
# ============================================================
if = {
    limit = { check_variable = { which = inOrbit value = @mpp_yes } }

    # ------------------------------------------------------------
    # Subcaso 1A: Ángulo definido explícitamente
    # ------------------------------------------------------------
    if = {
        limit = {
            NOT = {
                check_variable = { which = tAngle value = @mpp_no }
                check_variable = { which = tAngle value = @mpp_random }
            }
        }
        create_ambient_object = {
            type = $TYPE$
            $T_LOCATION$                            # Target location
            entity_offset = $LOCATION_DISTANCE$
            entity_offset_angle = $LOCATION_ANGLE$
            entity_offset_height = -10              # Ajuste de altura para objetos orbitales
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en $T_LOCATION$ a distancia $LOCATION_DISTANCE$ con ángulo $LOCATION_ANGLE$ (inOrbit)"
        }
    }

    # ------------------------------------------------------------
    # Subcaso 1B: Ángulo aleatorio
    # ------------------------------------------------------------
    else_if = {
        limit = {
            check_variable = { which = tAngle value = @mpp_random }
        }
        create_ambient_object = {
            type = $TYPE$
            $T_LOCATION$
            entity_offset = $LOCATION_DISTANCE$
            entity_offset_angle = random
            entity_offset_height = -10
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en $T_LOCATION$ a distancia $LOCATION_DISTANCE$ con ángulo random (inOrbit)"
        }
    }

    # ------------------------------------------------------------
    # Subcaso 1C: Rango de ángulos definido (min/max)
    # ------------------------------------------------------------
    else = {
        create_ambient_object = {
            type = $TYPE$
            $T_LOCATION$
            entity_offset = $LOCATION_DISTANCE$
            entity_offset_angle = {
                min = $LOCATION_MIN_ANGLE$
                max = $LOCATION_MAX_ANGLE$
            }
            entity_offset_height = -10
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en $T_LOCATION$ a distancia $LOCATION_DISTANCE$ con ángulo ($LOCATION_MIN_ANGLE$-$LOCATION_MAX_ANGLE$) (inOrbit)"
        }
    }
}

# ============================================================
# CASO 2: El objeto NO está en órbita
# ============================================================
else = {
    # ------------------------------------------------------------
    # Subcaso 2A: Ángulo definido explícitamente
    # ------------------------------------------------------------
    if = {
        limit = {
            NOT = {
                check_variable = { which = tAngle value = @mpp_no }
                check_variable = { which = tAngle value = @mpp_random }
            }
        }
        create_ambient_object = {
            type = $TYPE$
            $T_LOCATION$
            entity_offset = $LOCATION_DISTANCE$
            entity_offset_angle = $LOCATION_ANGLE$
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en $T_LOCATION$ a distancia $LOCATION_DISTANCE$ con ángulo $LOCATION_ANGLE$"
        }
    }

    # ------------------------------------------------------------
    # Subcaso 2B: Ángulo aleatorio
    # ------------------------------------------------------------
    else_if = {
        limit = {
            check_variable = { which = tAngle value = @mpp_random }
        }
        create_ambient_object = {
            type = $TYPE$
            $T_LOCATION$
            entity_offset = $LOCATION_DISTANCE$
            entity_offset_angle = random
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en $T_LOCATION$ a distancia $LOCATION_DISTANCE$ con ángulo random"
        }
    }

    # ------------------------------------------------------------
    # Subcaso 2C: Rango de ángulos definido (min/max)
    # ------------------------------------------------------------
    else = {
        create_ambient_object = {
            type = $TYPE$
            $T_LOCATION$
            entity_offset = $LOCATION_DISTANCE$
            entity_offset_angle = {
                min = $LOCATION_MIN_ANGLE$
                max = $LOCATION_MAX_ANGLE$
            }
        }
        # ---------- Debug ----------
        mpp_log_info_ao = {
            EFFECT = $EFFECT$ TEXT = "create_ambient_object $TYPE$ en $T_LOCATION$ a distancia $LOCATION_DISTANCE$ con ángulo ($LOCATION_MIN_ANGLE$-$LOCATION_MAX_ANGLE$)"
        }
    }
}

# ---------- Limpieza de variables ----------
clear_variable = inOrbit
clear_variable = tAngle
