limit = {
    check_variable = { which = tOrbitOffset value = $O_OFFSET$ }
}

spawn_planet = {
    class = $TYPE$
    orbit_distance = $F_ORBIT$
    size = $SIZE$
    orbit_angle = {
        min = $MIN_ANGLE$
        max = $MAX_ANGLE$
    }
    init_effect = {
        save_event_target_as = tPlanet
    }
}

event_target:tPlanet = {
    set_variable = { which = pOrbit value = $F_ORBIT$ }
}

# ---------- Debug ----------
mpp_log_info_go = { EFFECT = $EFFECT$ TEXT = "spawn_planet tipo $TYPE$ a $F_ORBIT$ tamaño $SIZE$ órbita $F_ORBIT$ ángulo ($MIN_ANGLE$-$MAX_ANGLE$)" }
