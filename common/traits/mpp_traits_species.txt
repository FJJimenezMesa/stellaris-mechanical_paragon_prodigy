# ================================
# 🧬 Species Trait: mpp_trait_nano_vector
# "Infección de nanorobots"
# ================================
# Descripción narrativa + lore:
#   Un vector biológico que transporta y mantiene nanorobots en la sangre. Estos nanites, conocidos como "sangrenaníticos", fueron originalmente desarrollados y liberados por el líder Parangón para asegurar su legado genético y capacidad de influencia incluso a distancia.
#   Los individuos portadores de este trait poseen longevidad mejorada, resistencia natural y habilidades sinérgicas para la producción y la investigación.
#   Su sangre se convierte en un conducto activo, un canal vivo que conecta su biología con la red de nanites del Parangón. El flujo de nanorobots es la base de una misteriosa herencia que impulsa la evolución de la especie hacia futuros estados biológicos avanzados.
# Descripción:
#   Un vector biológico que transporta y mantiene nanorobots en la sangre.
#   Funciona como portador activo de nanites ("sangrenanítica").
# Aplicación: Especies BIOLOGICAL, con posibilidad de cruzamiento avanzado.
# Enfoque: Rareza, longevidad, resiliencia y sinergias productivas.
# Afecta a: Pops y líderes biológicos.
# ================================
mpp_trait_nano_vector = {
    cost = 0
    initial = no
    randomized = no
    icon = "gfx/interface/icons/traits/mpp_trait_nano_vector.dds"

    opposites = { mpp_trait_nano_vector_null mpp_trait_nano_symbiosis }

    allowed_archetypes = { BIOLOGICAL MACHINE }

    species_potential_add = {
        AND = {
            OR = {
                from = {
                    mpp_is_biological_empire_no_gestal = yes
                    mpp_is_machine_empire_no_gestal = yes
                }
            }
            OR = {
                mpp_has_mpp_leader_with_vector_approved = yes
                mpp_has_mpp_leader_with_symbiosis_approved = yes
            }
        }
    }
    species_possible_remove = {
        always = no  # nunca se puede quitar manualmente
    }
    species_possible_merge_remove = {
        always = no  # nunca se puede quitar al fusionar especies
    }
    potential_crossbreeding_chance = @mpp_nano_vector_crossbreeding
    ai_weight = { @mpp_nano_vector_ai_weight }

    triggered_leader_modifier = {
        modifier = {
            mpp_leader_age = 50
        }
    }

    # --- Efectos condicionados ---
    triggered_pop_group_modifier = {
        potential = {
            species = { is_archetype = BIOLOGICAL }
        }

        modifier = {
            mpp_planet_population_bio_bonus = 1
            mpp_planet_job_production_bonus = 1
            mpp_planet_job_production_mult_bonus = 1
        }
    }

    triggered_pop_group_modifier = {
        potential = {
            species = { is_archetype = MACHINE }
        }
        modifier = {
            mpp_planet_population_robotic_bonus = 1
            mpp_planet_job_production_bonus = 1
            mpp_planet_job_production_mult_bonus = 1
        }
    }

    assembly_score = {
        modifier = {
            add = 0.5
        }
    }
}

# ================================
# 🧬 Species Trait: mpp_trait_nano_symbiosis
# "Parangón Nano-Simbiosis"
# ================================
# Descripción narrativa + lore:
#   Los individuos mecánicos de este trait han logrado establecer un control activo sobre los enjambres de nanites provenientes del líder Parangón. La simbiosis no es solo tecnológica, sino casi simbiótica: la máquina puede guiar, optimizar y sincronizar la acción de los nanorobots, aumentando su eficiencia en producción, investigación y liderazgo.
#   Estos nanites funcionan como una extensión de sus sistemas internos, permitiendo que la maquinaria adopte estrategias casi orgánicas de crecimiento y supervivencia. La influencia del Parangón sigue siendo la fuente, pero la máquina ahora la dirige de forma consciente y precisa, alcanzando un nivel de integración que los biológicos solo podrían soñar.
# Descripción:
#   Control activo sobre enjambres de nanites en cuerpos mecánicos.
#   "Operancy" implica control funcional directo sobre los nanites.
# Aplicación: Especies MACHINE
# Enfoque: Optimización mecánica, colonización eficiente y supremacía tecnológica.
# Afecta a: Pops y líderes mecánicos.
# ================================
mpp_trait_nano_symbiosis = {
    cost = 0
    initial = no
    randomized = no
    icon = "gfx/interface/icons/traits/mpp_trait_nano_symbiosis.dds"

    opposites = { mpp_trait_nano_vector mpp_trait_nano_vector_null }

    allowed_archetypes = { BIOLOGICAL MACHINE }

    species_potential_add = {
        AND = {
            OR = {
                from = {
                    mpp_is_biological_empire_no_gestal = yes
                    mpp_is_machine_empire_no_gestal = yes
                }
            }
            mpp_has_mpp_leader_with_symbiosis_approved = yes
        }
    }
    species_possible_remove = {
        always = no  # Nunca se puede quitar
    }
    species_possible_merge_remove = {
        always = no  # Tampoco al fusionar especies
    }
    potential_crossbreeding_chance = @mpp_nano_symbiosis_crossbreeding
    ai_weight = { @mpp_nano_symbiosis_ai_weight }  # La IA no lo usa

    triggered_leader_modifier = {
        mpp_leader_age = 50
    }

    # --- Efectos condicionados ---
    triggered_pop_group_modifier = {
        potential = {
            species = { is_archetype = BIOLOGICAL }
        }

        modifier = {
            mpp_planet_population_bio_bonus = 1
            mpp_planet_job_production_bonus = 1
            mpp_planet_job_production_mult_bonus = 1
            mpp_planet_job_research_production_bonus = 1
        }
    }

    triggered_pop_group_modifier = {
        potential = {
            species = { is_archetype = MACHINE }
        }
        mpp_planet_population_robotic_bonus = 1
        mpp_planet_job_production_bonus = 1
        mpp_planet_job_production_mult_bonus = 1
        mpp_planet_job_research_production_bonus = 1
    }

    slave_cost = {
        energy = 500
    }
    assembly_score = {
        modifier = {
            add = 0.5
        }
    }
}

# ================================
# 🧬 Species Trait: mpp_parangon_nano_operancy
# "Parangón Nano-Infusión"
# ================================
# Descripción narrativa + lore:
# Rasgo exclusivo para robots, los cuales carecen de sangre y no pueden establecer control directo sobre los nanorobots. En cambio, la infusión de nanites se integra parcialmente en sus sistemas, proporcionando aumentos de producción, eficiencia laboral y capacidad de investigación.
# Aunque la conexión es limitada, los nanites mantienen el legado del Parangón, sirviendo como un vínculo misterioso que permite a los robots recibir parte de su poder. La infusión nunca alcanza la simbiosis total de las máquinas, pero confiere una ventaja tangible y una presencia narrativa de la influencia del Parangón sobre toda la red mecánica del universo.
# Descripción:
#   Rasgo mecánico avanzado que mejora producción, felicidad, eficiencia de líderes y trabajos,
#   además de mejorar ensamblaje y definir costes de esclavos.
# Aplicación: Exclusivo para especies ROBOT
# Enfoque: Optimización mecánica, colonización eficiente y supremacía tecnológica.
# Afecta a: Pops y líderes robóticos.
# ================================
mpp_trait_nano_operancy = {
    cost = 0
    initial = no
    randomized = no
    icon = "gfx/interface/icons/traits/mpp_parangon_nano_operancy.dds"

    opposites = { mpp_trait_nano_operancy_null }  # Excluyente con trait nulo

    allowed_archetypes = { ROBOT }

    species_potential_add = {
        OR = {
            mpp_has_mpp_leader_with_vector_approved = yes
            mpp_has_mpp_leader_with_symbiosis_approved = yes
        }
    }
    species_possible_remove = {
        always = no  # Nunca se puede quitar
    }
    species_possible_merge_remove = {
        always = no  # Tampoco al fusionar especies
    }
    potential_crossbreeding_chance = @mpp_nano_operancy_crossbreeding
    ai_weight = { @mpp_nano_operancy_ai_weight }  # No lo usa la IA

    assembly_score = {
        modifier = {
            add = 0.5
        }
        modifier = {
            add = 1.5
            from = { has_mining_designation = yes }
        }
        modifier = {
            add = 0.5
            from = { has_rural_designation = yes }
        }
    }

    triggered_pop_group_modifier = {
        potential = {
            species = { is_archetype = ROBOT }
        }

        modifier = {
            mpp_leader_species_bonus = 1
            mpp_planet_population_bonus = 1
            mpp_planet_job_production_bonus = 1
            mpp_planet_job_research_production_bonus = 1
        }

    }
}
