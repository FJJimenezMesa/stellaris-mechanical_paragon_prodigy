# Nomenclaturas bio
#   Vessel → marcado por nanos en sangre.
#   Hollow / Null → hueco, no puede llenarse con la “sangre/nanoherencia”.
#   Disciple → coincide clase
#   Acolyte → no coincide clase
# Ramas de evolución bio
#   Genesis / Genetic
#   Syntesis / Synthetic
#   Unity / Full-Mechanic 
# Nomenclaturas robot
#   Prime Engine → Es robot con nanos
#   Static Frame → Es robot sin nanos
#   Mimic → coincide clase
#   Module? → no coincide clase
# Marcadores finales de ramas nivel+
#   Apex → nivel+ con nanos
#   Paragon → nivel++ con nanos (quedaría Apex Paragon)
#   Vanguard → nivel+ sin con nanos

# Ruler

mpp_ruler_trait_nano_vector = {
    cost = 1
    starting_ruler_trait = yes
    force_councilor_trait = yes                     # Se considera un consejero
    initial = no                                    # No aparece inicialmente
    randomized = no                                 # No se asigna aleatoriamente
    immortal_leaders = yes

    inline_script = {
        script = trait/icon
        CLASS = leader
        ICON = GFX_mpp_ruler_trait_nano_vector
        RARITY = paragon
        COUNCIL = yes
        TIER = 3
    }

    # Clases de líder a las que aplica
    leader_class = { governor admiral scientist general }

    leader_potential_add = {
        # Solo el líder del imperio puede recibir este trait
        is_ruler = yes
        # Puede ser líder biológico o máquina
        OR = {
            owner = { species = { is_archetype = BIOLOGICAL } }
            owner = { species = { is_archetype = MACHINE } }
        }
    }

    requires_traits = {
        mpp_trait_nano_vector       # Solo si líder tiene el trait de especies nano_vector
    }

    opposites = { mpp_ruler_trait_nano_symbiosis }

    #	self_modifier = { }					Modifier applied to the leader itself.
    #	modifier = { }						Modifier applied to scope leader is assigned to (e.g. fleet, or science ship) or country if councilor trait.
    #	planet_modifier = { }				Modifier applied to planet if General assigned to defense army.
    #	system_modifier = { }				Modifier applied to system if General or Governor is assigned to planet within it.
    #
    #	triggered_modifier = { }			Modifier applied to location the leader is assigned to if the given condition is fullfilled. THIS = Leader
    #	triggered_self_modifier = { }		Modifier applied to the leader itself if the given condition is fullfilled. THIS = Leader
    #	triggered_planet_modifier = { }		Modifier applied to the planet on which General is assigned to defend if the given condition is fullfilled. THIS = Leader

    # Modificadores aplicados así mismo
    self_modifier = {
        mpp_ruler_self = 1
    }

    # Modificadores aplicados así mismo
    self_modifier = {
        mpp_ruler_self = 1
    }

    # Modificadores aplicados al scope de líder (e.g. fleet, or science ship)
    modifier = {
        mpp_ruler = 1
    }

    # Bonificaciones específicas por clase de líder
    triggered_modifier = {
        potential = { leader_class = scientist }
        modifier = {
            mpp_ruler_scientist = 1
        }
    }

    triggered_modifier = {
        potential = { leader_class = admiral }
        modifier = {
            mpp_ruler_admiral = 1
        }
    }

    triggered_modifier = {
        potential = { leader_class = general }
        modifier = {
            mpp_ruler_general = 1
        }
    }

    triggered_modifier = {
        potential = { leader_class = governor }
        modifier = {
            mpp_ruler_governor = 1
        }
    }
}
