mpp_is_planet_scope = {
    AND = {
        is_scope_valid = yes
        is_scope_type = planet
    }
}
mpp_is_galactic_object = {
    AND = {
        is_scope_valid = yes
        is_scope_type = galactic_object
    }
}

mpp_is_planet_class_asteroid = {
    OR = {
        is_planet_class = pc_asteroid
        is_planet_class = pc_ice_asteroid
        is_planet_class = pc_crystal_asteroid
        is_planet_class = pc_rare_crystal_asteroid
    }
}

mpp_target_is_planet = {
    [[TARGET]
        if = {
            limit = { exists = event_target:$TARGET$ }

            event_target:$TARGET$ = {
                if = {
                    limit = {
                        OR = {
                            mpp_is_planet_class_planet = yes
                            mpp_is_planet_class_asteroid = yes
                        }
                    }
                    always = yes
                }
                else = { always = no }
            }
        }
        else = { always = no }
    ]

    [[TARGET]
        always = no
    ]
}
