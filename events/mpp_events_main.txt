# BEGIN_INFO
# info_type      : event
# id             : mpp_events_master.10
# name           : Evento de asignación de traits a pops biológicos
# type           : master
# target         : pops biológicos
# description    : Aplica traits mpp_trait_nano_vector o mpp_trait_nano_symbiosis a pops biológicos según el líder aprobado del planeta.
# related_events : []
# notes          :
#   • Se dispara solo si el pop es biológico
#   • Considera el tipo de líder activo (Vector o Symbiosis)
# END_INFO
planet_event = {
    id = mpp_events_master.10
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        fromfrom = {
            # Pop scope
            species = {
                is_archetype = BIOLOGICAL
            }
            owner = {
                # Country scope
                OR = {
                    mpp_has_mpp_leader_with_vector_approved = yes
                    mpp_has_mpp_leader_with_symbiosis_approved = yes
                }
            }
        }
    }

    immediate = {
        fromfrom = {
            # Pop scope
            if = {
                limit = {
                    owner = {
                        mpp_has_mpp_leader_with_vector_approved = yes
                    }
                }
                random_list = {
                    1 = {
                        modifier = {
                            factor = @mpp_nano_vector_bio_vector_pop_chance
                        }
                        mpp_add_species_trait = { TRAIT = mpp_trait_nano_vector }

                        modifier = {
                            factor = @mpp_nano_vector_bio_null_pop_chance
                        }
                        mpp_add_species_trait = { TRAIT = mpp_trait_nano_vector_null }
                    }
                }
            }
            else_if = {
                limit = {
                    owner = {
                        mpp_has_mpp_leader_with_symbiosis_approved = yes

# related_events : mpp_events_master.20, mpp_events_master.21
# notes          :
#   • Se puede disparar tanto al añadir líderes como al subir de nivel
#   • Garantiza coherencia de traits con las reglas de Vector y Symbiosis
# END_INFO
country_event = {
    id = mpp_events_master.22
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        from = {
            OR = {
                species = {
                    is_archetype = BIOLOGICAL
                    is_archetype = MACHINE
                }
            }
            OR = {
                owner = {
                    # Country scope
                    OR = {
                        mpp_has_mpp_leader_with_vector_approved = yes
                        mpp_has_mpp_leader_with_symbiosis_approved = yes
                    }
                }
                ruler = {
                    # Ruler scope
                    OR = {
                        has_trait = mpp_trait_nano_vector
                        has_trait = mpp_trait_nano_symbiosis
                    }
                }
            }
        }
    }

    immediate = {
        from = {
            remove_trait = mpp_leader_trait_bio_disciple
            remove_trait = mpp_leader_trait_bio_vessel_disciple
            remove_trait = mpp_leader_trait_bio_vanguard_vessel_disciple
            remove_trait = mpp_leader_trait_bio_null_disciple
            remove_trait = mpp_leader_trait_bio_acolyte
            remove_trait = mpp_leader_trait_bio_vessel_acolyte
            remove_trait = mpp_leader_trait_bio_vanguard_vessel_acolyte
            remove_trait = mpp_leader_trait_bio_null_acolyte

            remove_trait = mpp_leader_trait_machine_mimic
            remove_trait = mpp_leader_trait_machine_prime_engine_mimic
            remove_trait = mpp_leader_trait_machine_apex_prime_engine_mimic
            remove_trait = mpp_leader_trait_machine_null_mimic
            remove_trait = mpp_leader_trait_machine_module
            remove_trait = mpp_leader_trait_machine_prime_engine_module
            remove_trait = mpp_leader_trait_machine_apex_prime_engine_module
            remove_trait = mpp_leader_trait_machine_null_module

            if = {
                limit = {
                    species = {
                        is_archetype = BIOLOGICAL
                    }
                }
                if = {
                    limit = {
                        has_trait = mpp_trait_nano_vector
                    }
                    if = {
                        limit = { mpp_ruler_and_leader_same_class = yes }
                        mpp_add_leader_trait = { TRAIT = mpp_leader_trait_bio_vessel_disciple }
                    }
                    else_if = {
                        limit = { mpp_ruler_and_leader_not_same_class = yes }
                        mpp_add_leader_trait = { TRAIT = mpp_leader_trait_bio_vessel_acolyte }
                    }
                }
                else_if = {
                    limit = {
                        has_trait = mpp_trait_nano_symbiosis
                    }
                    if = {
                        limit = { mpp_ruler_and_leader_same_class = yes }
                        mpp_add_leader_trait = { TRAIT = mpp_leader_trait_bio_vanguard_vessel_disciple }
                    }
                    else_if = {
                        limit = { mpp_ruler_and_leader_not_same_class = yes }
                        mpp_add_leader_trait = { TRAIT = mpp_leader_trait_bio_vanguard_vessel_acolyte }
                    }
                }
                else_if = {
                    limit = {
                        has_trait = mpp_trait_nano_vector_null
                    }
                    if = {
                        limit = { mpp_ruler_and_leader_same_class = yes }
                        mpp_add_leader_trait = { TRAIT = mpp_leader_trait_bio_null_disciple }
                    }
